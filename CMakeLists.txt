cmake_minimum_required(VERSION 3.10)
project(UltrasoundVideoTracking CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenCV REQUIRED)
add_executable(track_ultrasound cpp/track_ultrasound.cpp)
target_include_directories(track_ultrasound PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(track_ultrasound ${OpenCV_LIBS})
target_compile_options(track_ultrasound PRIVATE -Wall -Wextra)
